DROP TABLE IF EXISTS backup_history;
DROP TABLE IF EXISTS profile;
DROP TABLE IF EXISTS system_constants;

CREATE TABLE system_constants (
  id                  INTEGER      GENERATED BY DEFAULT AS IDENTITY
 ,constant_name       TEXT         NOT NULL
 ,constant_type       TEXT         CONSTRAINT const_type CHECK (constant_type in ('text' , 'numeric' , 'boolean' , 'timestamp'))
 ,constant_value      TEXT         NOT NULL
 ,remarks             TEXT
 ,user_id             UUID         NOT NULL
                                   REFERENCES auth.users (id)
 ,PRIMARY KEY (constant_name, user_id)
);

CREATE TABLE profile (
  id                  INTEGER      GENERATED BY DEFAULT AS IDENTITY
 ,user_id             UUID         PRIMARY KEY
 ,user_name           TEXT         NOT NULL
 ,profile_icon        BOOLEAN
);

CREATE TABLE backup_history (
  id                  INTEGER      GENERATED BY DEFAULT AS IDENTITY
                                   PRIMARY KEY
 ,data_name           TEXT         NOT NULL
 ,backup_date         TIMESTAMP WITH TIME ZONE  NOT NULL
 ,user_id             UUID         NOT NULL
                                   REFERENCES auth.users (id)
);
