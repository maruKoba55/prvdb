■ログインユーザーのみに操作を許可

drop policy if exists "system_constants insert : only own user_id" on system_constants;
drop policy if exists "system_constants delete : only own user_id" on system_constants;
drop policy if exists "system_constants update : only own user_id" on system_constants;
drop policy if exists "system_constants select : only own user_id" on system_constants;

create policy "system_constants insert : only own user_id"
on "public"."system_constants"
as PERMISSIVE
for INSERT
to public
with check (
  (auth.uid() = user_id)
);
create policy "system_constants delete : only own user_id"
on "public"."system_constants"
as PERMISSIVE
for DELETE
to public
using (
  (auth.uid() = user_id)
);
create policy "system_constants update : only own user_id"
on "public"."system_constants"
as PERMISSIVE
for UPDATE
to public
using (
  (auth.uid() = user_id)
);
create policy "system_constants select : only own user_id"
on "public"."system_constants"
as PERMISSIVE
for SELECT
to public
using (
  (auth.uid() = user_id)
);
